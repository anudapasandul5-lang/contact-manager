# UI Revamp — Clean & Minimal Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the clay neumorphism design system with a clean, minimal aesthetic (Linear/Notion style) across the entire Contact Manager app on the `ui-revamp` branch.

**Architecture:** Hybrid approach — CSS variable swap in `globals.css` updates Tailwind utilities automatically; targeted JSX rewrites update node components and contacts page which use hardcoded inline `style={}` props that bypass CSS vars. One new file (`ContactTable.tsx`) is created; `ContactsGrid.tsx` gains a view-mode toggle.

**Tech Stack:** Next.js 16 App Router, Tailwind CSS v4, @xyflow/react 12, shadcn/ui (Base UI primitives), Lucide React

```tsx
"use client";

import Link from "next/link";
import { usePathname, useRouter } from "next/navigation";
import { useTransition, useState, useRef, useEffect } from "react";
import { Sun, Moon, LogOut, Plus, User, Building2, Truck, FolderKanban, ChevronDown } from "lucide-react";
import { cn } from "@/lib/utils";
import { useTheme } from "@/hooks/useTheme";
import { ContactModal } from "./ContactModal";
import { CompanyModal } from "./CompanyModal";
import { ProjectModal } from "./ProjectModal";
import { VendorModal } from "./VendorModal";

const tabs = [
  { href: "/mind-map", label: "Mind Map" },
  { href: "/contacts", label: "Contacts" },
];

const NEW_ITEMS = [
  { label: "Contact", icon: User, action: "contact" as const },
  { label: "Company", icon: Building2, action: "company" as const },
  { label: "Vendor", icon: Truck, action: "vendor" as const },
  { label: "Project", icon: FolderKanban, action: "project" as const },
];

export function Header() {
  const pathname = usePathname();
  const router = useRouter();
  const [isPending, startTransition] = useTransition();
  const { theme, toggleTheme, mounted } = useTheme();
  const isDark = mounted && theme === "dark";

  const [newMenuOpen, setNewMenuOpen] = useState(false);
  const [contactOpen, setContactOpen] = useState(false);
  const [companyOpen, setCompanyOpen] = useState(false);
  const [vendorOpen, setVendorOpen] = useState(false);
  const [projectOpen, setProjectOpen] = useState(false);
  const newMenuRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    function handleClickOutside(e: MouseEvent) {
      if (newMenuRef.current && !newMenuRef.current.contains(e.target as Node)) {
        setNewMenuOpen(false);
      }
    }
    if (newMenuOpen) document.addEventListener("mousedown", handleClickOutside);
    return () => document.removeEventListener("mousedown", handleClickOutside);
  }, [newMenuOpen]);

  function openModal(action: "contact" | "company" | "vendor" | "project") {
    setNewMenuOpen(false);
    if (action === "contact") setContactOpen(true);
    if (action === "company") setCompanyOpen(true);
    if (action === "vendor") setVendorOpen(true);
    if (action === "project") setProjectOpen(true);
  }

  async function handleSignOut() {
    await fetch("/api/auth/sign-out", { method: "POST" });
    startTransition(() => {
      router.replace("/login");
      router.refresh();
    });
  }

  return (
    <>
      <header className="sticky top-0 z-40 flex h-14 items-center border-b border-border bg-card px-6">
        {/* Brand */}
        <div className="flex items-center gap-2.5 mr-8">
          <div className="h-2.5 w-2.5 rounded-sm bg-primary" />
          <span className="text-sm font-semibold text-foreground">Contact Manager</span>
        </div>

        {/* Nav tabs */}
        <nav className="flex items-end h-full gap-1">
          {tabs.map((tab) => {
            const isActive = pathname.startsWith(tab.href);
            return (
              <Link
                key={tab.href}
                href={tab.href}
                className={cn(
                  "flex items-center px-3 h-full text-sm transition-colors border-b-2",
                  isActive
                    ? "border-primary text-foreground font-medium"
                    : "border-transparent text-muted-foreground hover:text-foreground"
                )}
              >
                {tab.label}
              </Link>
            );
          })}
        </nav>

        {/* Right side actions */}
        <div className="ml-auto flex items-center gap-2">
          {/* + New dropdown */}
          <div className="relative" ref={newMenuRef}>
            <button
              type="button"
              onClick={() => setNewMenuOpen((v) => !v)}
              className="flex items-center gap-1.5 rounded-md bg-primary px-3 py-1.5 text-[13px] font-medium text-primary-foreground hover:bg-primary/90 transition-colors"
            >
              <Plus className="h-3.5 w-3.5" />
              New
              <ChevronDown className="h-3 w-3 opacity-70" />
            </button>
            {newMenuOpen && (
              <div className="absolute right-0 top-full mt-1.5 z-50 w-40 rounded-lg border border-border bg-card shadow-md py-1">
                {NEW_ITEMS.map(({ label, icon: Icon, action }) => (
                  <button
                    key={action}
                    type="button"
                    onClick={() => openModal(action)}
                    className="flex w-full items-center gap-2.5 px-3 py-2 text-sm text-foreground hover:bg-muted transition-colors"
                  >
                    <Icon className="h-3.5 w-3.5 text-muted-foreground" />
                    {label}
                  </button>
                ))}
              </div>
            )}
          </div>

          {/* Dark mode toggle */}
          <button
            type="button"
            onClick={toggleTheme}
            className="flex h-8 w-8 items-center justify-center rounded-md text-muted-foreground hover:bg-muted hover:text-foreground transition-colors"
            title={mounted ? (isDark ? "Switch to light mode" : "Switch to dark mode") : "Toggle theme"}
          >
            {isDark ? <Sun className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
          </button>

          {/* Sign out */}
          <button
            type="button"
            onClick={handleSignOut}
            disabled={isPending}
            className="flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground transition-colors disabled:opacity-60"
          >
            <LogOut className="h-3.5 w-3.5" />
            {isPending ? "Signing out…" : "Sign out"}
          </button>
        </div>
      </header>

      <ContactModal open={contactOpen} onOpenChange={setContactOpen} onSaved={() => window.dispatchEvent(new CustomEvent("contact-manager:data-changed"))} />
      <CompanyModal open={companyOpen} onOpenChange={setCompanyOpen} onSaved={() => window.dispatchEvent(new CustomEvent("contact-manager:data-changed"))} />
      <VendorModal open={vendorOpen} onOpenChange={setVendorOpen} onSaved={() => window.dispatchEvent(new CustomEvent("contact-manager:data-changed"))} />
      <ProjectModal open={projectOpen} onOpenChange={setProjectOpen} onSaved={() => window.dispatchEvent(new CustomEvent("contact-manager:data-changed"))} />
    </>
  );
}
```

```tsx
"use client";

import { useState } from "react";
import { ChevronUp, ChevronDown, Pencil, Trash2, MoreHorizontal } from "lucide-react";
import type { ContactWithRelations, VendorWithRelations } from "@/lib/supabase/types";
import type { DirectoryItem } from "./directory-items";

interface ContactTableProps {
  items: DirectoryItem[];
  onEditContact: (contact: ContactWithRelations) => void;
  onDeleteContact: (contact: ContactWithRelations) => void;
  onEditVendor: (vendor: VendorWithRelations) => void;
  onDeleteVendor: (vendor: VendorWithRelations) => void;
}

type SortKey = "name" | "role" | "company" | "type";
type SortDir = "asc" | "desc";

function getInitials(name: string) {
  return name.split(" ").map((w) => w[0]).join("").toUpperCase().slice(0, 2);
}

function ContactRow({
  item,
  onEdit,
  onDelete,
}: {
  item: DirectoryItem;
  onEdit: () => void;
  onDelete: () => void;
}) {
  const [actionsOpen, setActionsOpen] = useState(false);

  const name = item.kind === "contact" ? item.contact.name : item.vendor.name;
  const role = item.kind === "contact" ? (item.contact.role ?? "—") : (item.vendor.specialty ?? "Vendor");
  const type = item.kind === "contact" ? item.contact.type : "vendor";
  const dotColor = type === "employee" ? "#22c55e" : "#f97316";
  const typeLabel = type === "employee" ? "Employee" : "Vendor";
  const initials = getInitials(name);

  const companies =
    item.kind === "contact"
      ? (item.contact.contact_companies ?? []).map((cc) => cc.companies?.name).filter(Boolean)
      : [];

  return (
    <tr className="border-b border-border hover:bg-muted/40 transition-colors">
      {/* Avatar + Name */}
      <td className="px-4 py-3">
        <div className="flex items-center gap-3">
          <div
            className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full text-[11px] font-bold text-white"
            style={{ background: dotColor }}

        </div>
      </td>

      {/* Role */}
      <td className="px-4 py-3 text-sm text-muted-foreground">{role}</td>

      {/* Company */}
      <td className="px-4 py-3 text-sm text-muted-foreground">

      </td>

      {/* Type badge */}
      <td className="px-4 py-3">
        <span
          className="inline-flex items-center gap-1.5 rounded-md px-2 py-0.5 text-[11px] font-semibold"
          style={{
            background: type === "employee" ? "#dcfce7" : "#ffedd5",
            color: type === "employee" ? "#15803d" : "#c2410c",
          }}
        >
          <div className="h-1.5 w-1.5 rounded-full" style={{ background: dotColor }} />

        </span>
      </td>

      {/* Actions */}
      <td className="px-4 py-3">
        <div className="relative flex justify-end">
          <button
            type="button"
            onClick={() => setActionsOpen((v) => !v)}

          >

              </button>

function SortHeader({
  label,
  sortKey,
  currentKey,
  dir,
  onSort,
}: {
  label: string;
  sortKey: SortKey;
  currentKey: SortKey;
  dir: SortDir;
  onSort: (key: SortKey) => void;
}) {
  const isActive = currentKey === sortKey;
  return (
    <th
      className="px-4 py-2.5 text-left text-[11px] font-semibold uppercase tracking-wide text-muted-foreground cursor-pointer hover:text-foreground transition-colors select-none"
      onClick={() => onSort(sortKey)}
    >
      <div className="flex items-center gap-1">
        {label}
        {isActive ? (
          dir === "asc" ? <ChevronUp className="h-3 w-3" /> : <ChevronDown className="h-3 w-3" />
        ) : (
          <ChevronDown className="h-3 w-3 opacity-0 group-hover:opacity-40" />
        )}
      </div>
    </th>
  );
}

export function ContactTable({
  items,
  onEditContact,
  onDeleteContact,
  onEditVendor,
  onDeleteVendor,
}: ContactTableProps) {
  const [sortKey, setSortKey] = useState<SortKey>("name");
  const [sortDir, setSortDir] = useState<SortDir>("asc");

  function handleSort(key: SortKey) {
    if (key === sortKey) {
      setSortDir((d) => (d === "asc" ? "desc" : "asc"));
    } else {
      setSortKey(key);
      setSortDir("asc");
    }
  }

  const sorted = [...items].sort((a, b) => {
    const aName = a.kind === "contact" ? a.contact.name : a.vendor.name;
    const bName = b.kind === "contact" ? b.contact.name : b.vendor.name;
    const aRole = a.kind === "contact" ? (a.contact.role ?? "") : (a.vendor.specialty ?? "");
    const bRole = b.kind === "contact" ? (b.contact.role ?? "") : (b.vendor.specialty ?? "");
    const aType = a.kind === "contact" ? a.contact.type : "vendor";
    const bType = b.kind === "contact" ? b.contact.type : "vendor";

    let valA = "";
    let valB = "";
    if (sortKey === "name") { valA = aName; valB = bName; }
    else if (sortKey === "role") { valA = aRole; valB = bRole; }
    else if (sortKey === "type") { valA = aType; valB = bType; }

    const cmp = valA.localeCompare(valB);
    return sortDir === "asc" ? cmp : -cmp;
  });

  return (
    <div className="rounded-lg border border-border overflow-hidden">
      <table className="w-full text-sm">
        <thead className="border-b border-border bg-muted/50">
          <tr>
            <SortHeader label="Name" sortKey="name" currentKey={sortKey} dir={sortDir} onSort={handleSort} />
            <SortHeader label="Role" sortKey="role" currentKey={sortKey} dir={sortDir} onSort={handleSort} />
            <th className="px-4 py-2.5 text-left text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Company</th>
            <SortHeader label="Type" sortKey="type" currentKey={sortKey} dir={sortDir} onSort={handleSort} />
            <th className="px-4 py-2.5" />
          </tr>
        </thead>
        <tbody>
          {sorted.map((item) => (
            <ContactRow
              key={item.key}
              item={item}
              onEdit={
                item.kind === "contact"
                  ? () => onEditContact(item.contact)
                  : () => onEditVendor(item.vendor)
              }
              onDelete={
                item.kind === "contact"
                  ? () => onDeleteContact(item.contact)
                  : () => onDeleteVendor(item.vendor)
              }
            />
          ))}
        </tbody>
      </table>
    </div>
  );
}
```
