```sql
-- Auto-update updated_at timestamp triggers
-- Run this in Supabase SQL Editor: Dashboard -> SQL Editor -> New Query

-- 1. Create a reusable trigger function
CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 2. Trigger for person_relationships
DROP TRIGGER IF EXISTS trg_person_relationships_updated_at ON person_relationships;
CREATE TRIGGER trg_person_relationships_updated_at
  BEFORE UPDATE ON person_relationships
  FOR EACH ROW
  EXECUTE FUNCTION set_updated_at();

-- 3. Trigger for intro_requests
DROP TRIGGER IF EXISTS trg_intro_requests_updated_at ON intro_requests;
CREATE TRIGGER trg_intro_requests_updated_at
  BEFORE UPDATE ON intro_requests
  FOR EACH ROW
  EXECUTE FUNCTION set_updated_at();
```

The function uses `CREATE OR REPLACE` so it is idempotent. The triggers use `DROP TRIGGER IF EXISTS` before `CREATE TRIGGER` so re-running the script is safe.
