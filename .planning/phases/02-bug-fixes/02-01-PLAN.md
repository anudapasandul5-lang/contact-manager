```typescript
  // Determine which projects have contacts linked
  const projectHasContacts = new Set<string>();
  data.contacts.forEach((contact) => {
    (contact.contact_projects ?? []).forEach((cp) => {
      projectHasContacts.add(cp.projects.id);
    });
  });

  const orphanProjectIds: string[] = [];

  data.projects.forEach((project) => {
    const nodeId = `project-${project.id}`;
    // Project has a company with a known position — place in orbit
    if (project.company_id) {
      const companyPos = positionMap.get(`company-${project.company_id}`);
      if (companyPos) {
        const angle = Math.atan2(companyPos.y, companyPos.x);
        positionMap.set(nodeId, {
          x: companyPos.x + Math.cos(angle) * PROJECT_ORBIT,
          y: companyPos.y + Math.sin(angle) * PROJECT_ORBIT,
        });
        return;
      }
    }
    // Project has contacts linked — place near center-ish ring (not orphaned visually)
    if (projectHasContacts.has(project.id)) {
      orphanProjectIds.push(nodeId);
      return;
    }
    // Truly orphaned — no company, no contacts
    orphanProjectIds.push(nodeId);
  });

  if (orphanProjectIds.length > 0) {
    placeOnRing(orphanProjectIds, 750);
  }
```
